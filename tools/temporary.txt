def read(path):
    return path.read_text(encoding="utf-8")


def strip_front_matter(text):
    if text.startswith("---\n"):
        parts = text.split("---\n", 2)
        if len(parts) == 3:
            return parts[2].strip()
    return text.strip()


def strip_review_gate(text):
    lines = text.splitlines()
    for index, line in enumerate(lines):
        if line.strip() in {"## Segment review gate", "## Placeholder review gate"}:
            return "\n".join(lines[:index]).rstrip()
    return text.strip()


def segment_number(path):
    match = re.match(r"(\d+)-", path.name)
    return int(match.group(1)) if match else 0


def load_segments(language):
    folder = ROOT / "localization" / "sources" / language / "between-potential-and-ideal"
    return {
        segment_number(path): strip_review_gate(strip_front_matter(read(path)))
        for path in sorted(folder.glob("*.md"))
    }


def assemble_main(language):
    base = ROOT / "localization" / "sources" / language / f"between-potential-and-ideal-{language}.md"
    text = read(base)
    marker = "Translation control note"
    cut = text.find(marker)
    if cut >= 0:
        heading = text.rfind("\n## ", 0, cut)
        text = text[:heading] if heading >= 0 else text[:cut]
    parts = [text.rstrip()]
    parts.extend(load_segments(language)[key] for key in sorted(load_segments(language)))
    return "\n\n".join(part.strip() for part in parts if part.strip()) + "\n"


def markdown_to_text(text):
    text = re.sub(r"!\[([^]]*)\]\([^)]*\)", r"\1", text)
    text = re.sub(r"\[([^]]+)\]\([^)]*\)", r"\1", text)
    text = re.sub(r"^#{1,6}\s*", "", text, flags=re.MULTILINE)
    text = re.sub(r"[*_`>]", "", text)
    return re.sub(r"\n{3,}", "\n\n", text).strip() + "\n"


def find_image(reference):
    name = Path(reference).name
    direct = SITE / "figures" / name
    if direct.is_file():
        return direct
    found = list(SITE.rglob(name))
    return found[0] if found else None


def make_docx(markdown_text, output):
    from docx import Document
    from docx.shared import Inches, Pt
    document = Document()
    styles = document.styles
    styles["Normal"].font.name = "DejaVu Sans"
    styles["Normal"].font.size = Pt(10.5)
    image_pattern = re.compile(r"!\[([^]]*)\]\(([^)]+)\)")
    for raw in markdown_text.splitlines():
        line = raw.strip()
        if not line:
            document.add_paragraph()
            continue
        image = image_pattern.fullmatch(line)
        if image:
            path = find_image(image.group(2))
            if path:
                try:
                    document.add_picture(str(path), width=Inches(6.2))
                except Exception:
                    document.add_paragraph(image.group(1))
            else:
                document.add_paragraph(image.group(1))
            continue
        heading = re.match(r"^(#{1,6})\s+(.+)$", line)
        if heading:
            document.add_heading(heading.group(2), level=min(len(heading.group(1)), 4))
        elif re.match(r"^[-*]\s+", line):
            document.add_paragraph(re.sub(r"^[-*]\s+", "", line), style="List Bullet")
        elif re.match(r"^\d+\.\s+", line):
            document.add_paragraph(re.sub(r"^\d+\.\s+", "", line), style="List Number")
        else:
            document.add_paragraph(re.sub(r"[*_`]", "", line))
    output.parent.mkdir(parents=True, exist_ok=True)
    document.save(output)


def make_pdf(docx_path, output_dir):
    subprocess.run([
        "libreoffice", "--headless", "--convert-to", "pdf", "--outdir", str(output_dir), str(docx_path)
    ], check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)


def document_html(title, language, markdown_text):
    import markdown
    body = markdown.markdown(markdown_text, extensions=["tables", "fenced_code", "sane_lists"])
    body = body.replace('src="../figures/', 'src="../../figures/')
    return f'''<!doctype html>
<html lang="{language}" dir="ltr"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">
<title>{html.escape(title)} · Between Potential and Ideal</title><meta name="robots" content="index,follow">
<style>body{{margin:0;background:#07090d;color:#e9edf5;font:17px/1.72 system-ui,sans-serif}}main{{max-width:900px;margin:auto;padding:48px 24px 90px}}h1,h2,h3,h4{{color:#f2c45e;line-height:1.2}}a{{color:#72d8d0}}img{{max-width:100%;height:auto;border-radius:18px}}blockquote{{border-left:3px solid #f2c45e;padding-left:18px;color:#cbd3df}}code{{white-space:pre-wrap}}</style></head><body><main>{body}</main></body></html>'''


def write_package(language, package, markdown_text):
    folder = SITE / "files" / language
    folder.mkdir(parents=True, exist_ok=True)
    stem = folder / f"{package}-{language}"
    source = markdown_text.strip() + "\n"
    stem.with_suffix(".md").write_text(source, encoding="utf-8")
    stem.with_suffix(".txt").write_text(markdown_to_text(source), encoding="utf-8")
    stem.with_suffix(".html").write_text(document_html(package.replace("-", " ").title(), language, source), encoding="utf-8")
    make_docx(source, stem.with_suffix(".docx"))
    make_pdf(stem.with_suffix(".docx"), folder)
